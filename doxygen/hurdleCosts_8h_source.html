37  static void hourForEachLink(IntermediateValues& iv, State& state)
38  {
39  if (state.link->useHurdlesCost)
40  {
41  const double flowLinear = state.ntc.ValeurDuFlux[state.link->index];
42 
43  if (state.link->useLoopFlow)
44  {
45  const double loopFlow = state.problemeHebdo->ValeursDeNTC[state.hourInTheWeek]
46  .ValeurDeLoopFlowOrigineVersExtremite[state.link->index];
47  if (flowLinear - loopFlow > 0.)
48  {
49  const double hurdleCostDirect = (flowLinear - loopFlow)
50  * state.link->parameters
51  .entry[Data::fhlHurdlesCostDirect]
52  [state.hourInTheYear];
53  iv.hour[state.hourInTheYear] += hurdleCostDirect;
54  state.annualSystemCost += hurdleCostDirect;
55  }
56  else
57  {
58  const double hurdleCostIndirect = -(flowLinear - loopFlow)
59  * state.link->parameters
60  .entry[Data::fhlHurdlesCostIndirect]
61  [state.hourInTheYear];
62  iv.hour[state.hourInTheYear] += hurdleCostIndirect;
63  state.annualSystemCost += hurdleCostIndirect;
64  }
65  }
66  else
67  {
68  if (flowLinear > 0.)
69  {
70  const double hurdleCostDirect = flowLinear
71  * state.link->parameters
72  .entry[Data::fhlHurdlesCostDirect]
73  [state.hourInTheYear];
74  iv.hour[state.hourInTheYear] += hurdleCostDirect;
75  state.annualSystemCost += hurdleCostDirect;
76  }
77  else
78  {
79  const double hurdleCostIndirect = -flowLinear
80  * state.link->parameters
81  .entry[Data::fhlHurdlesCostIndirect]
82  [state.hourInTheYear];
83  iv.hour[state.hourInTheYear] += hurdleCostIndirect;
84  state.annualSystemCost += hurdleCostIndirect;
