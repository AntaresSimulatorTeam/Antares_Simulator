40  static double calculateEnergyDeficitCosts(const State& state)
41  {
42  const auto hourInWeek = state.hourInTheWeek;
43  const auto& hourlyResults = *state.hourlyResults;
44 
45  const double unsuppliedCost = hourlyResults.ValeursHorairesDeDefaillancePositive[hourInWeek]
46  * state.area->thermal.unsuppliedEnergyCost;
47 
48  const double spilledCost = (hourlyResults.ValeursHorairesDeDefaillanceNegative[hourInWeek]
49  + state.resSpilled.entry[state.area->index][hourInWeek])
50  * state.area->thermal.spilledEnergyCost;
51 
52  return unsuppliedCost + spilledCost;
53  }
54 
55  static double calculateHydroCosts(const State& state)
56  {
57  const auto hourInWeek = state.hourInTheWeek;
58  const auto& hourlyResults = *state.hourlyResults;
59  const auto& hydroCharacteristics = state.problemeHebdo
60  ->CaracteristiquesHydrauliques[state.area->index];
61 
62  const double waterValueCost = hydroCharacteristics.WeeklyWaterValueStateRegular
63  * (hourlyResults.TurbinageHoraire[hourInWeek]
64  - state.area->hydro.pumpingEfficiency
65  * hourlyResults.PompageHoraire[hourInWeek]);
66 
67  const double storageReserveCost = state.reserveData
68  ? state.reserveData->at(state.area->index)
69  .STStorageClusterReserveParticipationCostForYear
70  [state.hourInTheYear]
71  : 0.0;
72 
73  const double hydroReserveCost = state.reserveData ? state.reserveData->at(state.area->index)
74  .HydroReserveParticipationCostForYear
75  [state.hourInTheYear]
76  : 0.0;
77 
78  return waterValueCost + storageReserveCost + hydroReserveCost;
79  }

81  static double calculateThermalCosts(const State& state)
82  {
83  double totalCost = 0.0;

90 
91  return totalCost;
92  }
