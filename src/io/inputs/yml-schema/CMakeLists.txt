set(SCHEMA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/schemas)
set(SCHEMA_FILES
        ${SCHEMA_DIR}/common.schema.yml
        ${SCHEMA_DIR}/model-library.schema.yml
        ${SCHEMA_DIR}/system.schema.yml
)

# Embed the schemas into the binary as byte arrays: no runtime file lookup, and no compiler limit
# on the length of string literals.
set(EMBEDDED_SCHEMAS_CPP ${CMAKE_CURRENT_BINARY_DIR}/embeddedSchemas.cpp)
set(EMBEDDED_SCHEMAS_CONTENT "// Generated by CMake from schemas/*.schema.yml. Do not edit.\n")
string(APPEND EMBEDDED_SCHEMAS_CONTENT "#include \"embeddedSchemas.h\"\n\n")
string(APPEND EMBEDDED_SCHEMAS_CONTENT "namespace Antares::IO::Inputs::YmlSchema\n{\n")
string(APPEND EMBEDDED_SCHEMAS_CONTENT "namespace\n{\n")
set(EMBEDDED_SCHEMAS_TABLE "")
set(schema_index 0)
foreach (schema_file ${SCHEMA_FILES})
    get_filename_component(schema_name ${schema_file} NAME)
    file(READ ${schema_file} schema_hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," schema_bytes "${schema_hex}")
    string(APPEND EMBEDDED_SCHEMAS_CONTENT
            "const unsigned char schema${schema_index}[] = {${schema_bytes}};\n")
    string(APPEND EMBEDDED_SCHEMAS_TABLE
            "    {\"${schema_name}\", {reinterpret_cast<const char*>(schema${schema_index}), sizeof(schema${schema_index})}},\n")
    math(EXPR schema_index "${schema_index} + 1")
    # Re-run CMake when a schema changes so that the embedded copy never goes stale
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${schema_file})
endforeach ()
string(APPEND EMBEDDED_SCHEMAS_CONTENT "} // namespace\n\n")
string(APPEND EMBEDDED_SCHEMAS_CONTENT
        "const std::vector<EmbeddedSchema>& embeddedSchemas()\n{\n")
string(APPEND EMBEDDED_SCHEMAS_CONTENT
        "    static const std::vector<EmbeddedSchema> schemas = {\n${EMBEDDED_SCHEMAS_TABLE}    };\n")
string(APPEND EMBEDDED_SCHEMAS_CONTENT "    return schemas;\n}\n")
string(APPEND EMBEDDED_SCHEMAS_CONTENT "} // namespace Antares::IO::Inputs::YmlSchema\n")
file(WRITE ${EMBEDDED_SCHEMAS_CPP} "${EMBEDDED_SCHEMAS_CONTENT}")

set(SOURCES
        schemaValidation.cpp
        embeddedSchemas.h
        ${EMBEDDED_SCHEMAS_CPP}
        include/antares/io/inputs/yml-schema/schemaValidation.h
)

add_library(yml-schema ${SOURCES})
add_library(Antares::yml-schema ALIAS yml-schema)

target_include_directories(yml-schema
        PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(yml-schema
        PUBLIC
        Antares::input-error
        yaml-cpp::yaml-cpp
        PRIVATE
        fmt::fmt
        jsoncons::jsoncons
)

install(DIRECTORY include/antares
        DESTINATION "include"
)
